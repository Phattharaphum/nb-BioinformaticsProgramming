ALGORITHM Ecoli_AMR_Genomic_Clinical_Prediction_Pipeline
INPUT:  clinical_csv_path, isolate_fasta_path, isolate_metadata_path, reference_amr_fasta_path, output_dir
OUTPUT: amr_predictions.csv, project2_evaluation_summary.json, evaluation_summary_dict
1. FUNCTION run_amr_prediction_pipeline(clinical_csv_path, isolate_fasta_path, isolate_metadata_path, reference_amr_fasta_path, output_dir):
     // [ลำดับที่ 1] เรียกใช้ Module 1 เพื่อสกัดโปรไฟล์ความเสี่ยงคลินิกและค่าสถิติ MIC จากข้อมูล 574,239 แถว
     clinical_profile <- CALL Module1.load_and_profile_clinical_resistance(clinical_csv_path)
       ├─ STREAM rows FROM clinical_csv_path WHERE drug_code IN ('AMC', 'AMP', 'CIP', 'NIT', 'TET')
       ├─ FOR EACH drug_code: COMPUTE S_rate, NS_rate, MIC50, MIC90, Geometric_Mean_MIC
       └─ FOR EACH isolate_ID: COUNT resistant drug_classes; IF count >= 3 THEN MARK AS Clinical_MDR
     // [ลำดับที่ 2] เรียกใช้ Module 2 เพื่ออ่านลำดับเบสจีโนม FASTA ของสายพันธุ์ E. coli และตรวจสอบคุณภาพลำดับเบส
     isolate_genomes <- CALL Module2.fetch_and_parse_isolate_genomes(isolate_fasta_path, isolate_metadata_path)
       ├─ fasta_records <- CALL Module2.parse_fasta_records(isolate_fasta_path)
       └─ FOR EACH record IN fasta_records:
            qc <- CALL Module2.compute_sequence_qc_metrics(record.sequence) // ตรวจความยาว, %GC, %N
            MERGE record WITH NCBI/ENA metadata AND qc INTO isolate_genomes
     // [ลำดับที่ 3] เรียกใช้ Module 3 เพื่อสแกนหายีนดื้อยาและตรวจหาการกลายพันธุ์ระดับกรดอะมิโน
     scan_results <- CALL Module3.scan_amr_genes_and_mutations(isolate_genomes, reference_amr_fasta_path)
       ├─ ref_genes <- CALL Module2.parse_fasta_records(reference_amr_fasta_path) // เรียกใช้ฟังก์ชันข้ามโมดูล
       └─ FOR EACH isolate IN isolate_genomes:
            FOR EACH acquired_gene IN ('blaTEM-1', 'blaCTX-M-15', 'blaOXA-1', 'ampC', 'qnrS1', 'tetA', 'tetB'):
              hit <- CALL Module3.align_and_score_gene_hit(isolate.sequence, ref_genes[acquired_gene], kmer=15)
              IF hit.identity >= 0.90 AND hit.coverage >= 0.85 THEN APPEND acquired_gene TO isolate.acquired_hits
            FOR EACH core_gene IN ('gyrA', 'parC', 'nfsA', 'nfsB'):
              aligned_cds <- CALL Module3.align_and_score_gene_hit(isolate.sequence, ref_genes[core_gene])
              aa_seq <- CALL Module3.translate_bacterial_cds(aligned_cds) // แปลรหัส Codon Table 11
              CHECK hotspot codons (gyrA 83,87; parC 80,84; nfsA 15; nfsB 44,94) AGAINST wild-type AA
     // [ลำดับที่ 4] เรียกใช้ Module 4 เพื่อแปลงยีนดื้อยาและการกลายพันธุ์เป็นคะแนนกลไกต่อยาปฏิชีวนะ 5 ชนิด
     mapped_profiles <- CALL Module4.map_genetic_determinants_to_drugs(scan_results, clinical_profile)
       └─ FOR EACH isolate_id, scan_info IN scan_results:
            FOR EACH drug_code IN ('AMC', 'AMP', 'CIP', 'NIT', 'TET'):
              score_info <- CALL Module4.evaluate_drug_mechanism_score(drug_code, scan_info.genes, scan_info.mutations, baseline_ns)
     // [ลำดับที่ 5] เรียกใช้ Module 5 เพื่อทำนายฟีโนไทป์ S/NS, คาดการณ์ค่า MIC, ระบุสายพันธุ์ MDR และประเมินผล
     evaluation_summary <- CALL Module5.predict_susceptibility_and_evaluate(mapped_profiles, clinical_profile, isolate_genomes, output_dir)
       ├─ FOR EACH isolate_id, drug_code:
       │    pred_mic <- wildtype_mic[drug_code] * score_info.estimated_fold_increase
       │    pred_phenotype <- 'NS' IF (pred_mic >= CLSI_ns_breakpoint[drug_code] OR prob_ns >= 0.50) ELSE 'S'
       ├─ IF count(NS_drug_classes of isolate) >= 3 THEN predicted_mdr <- True
       ├─ metrics <- CALL Module5.compute_confusion_and_concordance(predictions, actual_phenotypes)
       └─ WRITE amr_predictions.csv AND project2_evaluation_summary.json TO output_dir
     RETURN evaluation_summary
